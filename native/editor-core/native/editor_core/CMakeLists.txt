cmake_minimum_required(VERSION 3.16)
project(AstraEditorCore LANGUAGES CXX)

add_library(astra_editor_core SHARED editor_core.cpp)
target_compile_features(astra_editor_core PUBLIC cxx_std_17)
target_include_directories(astra_editor_core PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>)
target_compile_definitions(astra_editor_core PRIVATE ASTRA_EC_BUILD)
set_target_properties(astra_editor_core PROPERTIES
    CXX_EXTENSIONS OFF CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
if(MSVC)
    target_compile_options(astra_editor_core PRIVATE /W4 /permissive- /utf-8 /fp:strict)
elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(astra_editor_core PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    include(CTest)
endif()
if(BUILD_TESTING)
    add_executable(astra_editor_core_test editor_core_test.cpp)
    set_target_properties(astra_editor_core_test PROPERTIES CXX_EXTENSIONS OFF)
    target_link_libraries(astra_editor_core_test PRIVATE astra_editor_core)
    add_test(NAME astra_editor_core_test COMMAND astra_editor_core_test)
endif()

install(TARGETS astra_editor_core RUNTIME DESTINATION . LIBRARY DESTINATION . ARCHIVE DESTINATION lib)
install(FILES astra_editor_core.h astra_document.h DESTINATION include)

set(ASTRA_JSON_INCLUDE_DIR "" CACHE PATH "Official nlohmann/json 3.12.0 include directory")
if(NOT EXISTS "${ASTRA_JSON_INCLUDE_DIR}/nlohmann/json.hpp")
 message(FATAL_ERROR "Set ASTRA_JSON_INCLUDE_DIR to the verified official 3.12.0 include directory")
endif()
target_sources(astra_editor_core PRIVATE
    document.cpp timing_model.cpp structural_commands.cpp note_commands.cpp pulse_commands.cpp draft_commands.cpp)
target_include_directories(astra_editor_core PRIVATE ${ASTRA_JSON_INCLUDE_DIR})

# Keep the existing model differential/test ABI separate from the product ABI.
add_library(astra_editor_models SHARED timing_model.cpp structural_commands.cpp note_commands.cpp pulse_commands.cpp draft_commands.cpp tests/structural_test_abi.cpp)
target_compile_features(astra_editor_models PRIVATE cxx_std_17)
target_include_directories(astra_editor_models PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${ASTRA_JSON_INCLUDE_DIR})
if(MSVC)
 target_compile_options(astra_editor_models PRIVATE /utf-8 /W4 /fp:strict)
else()
 target_compile_options(astra_editor_models PRIVATE -ffp-contract=off)
endif()
add_executable(astra_timing_test tests/timing_tests.cpp timing_model.cpp structural_commands.cpp)
target_compile_features(astra_timing_test PRIVATE cxx_std_17)
target_include_directories(astra_timing_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${ASTRA_JSON_INCLUDE_DIR})
if(MSVC)
 target_compile_options(astra_timing_test PRIVATE /utf-8 /fp:strict /UNDEBUG)
endif()
add_test(NAME astra_timing_test COMMAND astra_timing_test)

add_executable(astra_pulse_differential pulse_differential_driver.cpp pulse_commands.cpp draft_commands.cpp timing_model.cpp structural_commands.cpp)
target_compile_features(astra_pulse_differential PRIVATE cxx_std_17)
target_include_directories(astra_pulse_differential PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${ASTRA_JSON_INCLUDE_DIR})
if(MSVC)
 target_compile_options(astra_pulse_differential PRIVATE /utf-8 /fp:strict)
endif()

add_executable(astra_note_semantic ../../tests/note_semantic_model.cpp note_commands.cpp timing_model.cpp structural_commands.cpp)
target_compile_features(astra_note_semantic PRIVATE cxx_std_17)
target_include_directories(astra_note_semantic PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${ASTRA_JSON_INCLUDE_DIR})
if(MSVC)
 target_compile_options(astra_note_semantic PRIVATE /utf-8 /fp:strict)
endif()
add_test(NAME astra_note_semantic COMMAND astra_note_semantic)
