cmake_minimum_required(VERSION 3.25)
project(AstraVstHost LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "This first-stage host supports Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
set(SDK_ROOT "${CMAKE_CURRENT_LIST_DIR}/../vendor/vst3sdk" CACHE PATH "Official VST3 SDK checkout")
add_subdirectory("${SDK_ROOT}" "${CMAKE_BINARY_DIR}/sdk" EXCLUDE_FROM_ALL)
add_library(astra_vst_host SHARED astra_vst_host.cpp
 "${SDK_ROOT}/public.sdk/source/vst/hosting/module_win32.cpp"
 "${SDK_ROOT}/public.sdk/source/vst/hosting/plugprovider.cpp"
 "${SDK_ROOT}/public.sdk/source/common/memorystream.cpp")
target_include_directories(astra_vst_host PRIVATE "${SDK_ROOT}")
target_compile_definitions(astra_vst_host PRIVATE NOMINMAX UNICODE _UNICODE)
target_compile_options(astra_vst_host PRIVATE /utf-8)
target_link_libraries(astra_vst_host PRIVATE sdk_hosting user32 ole32 shell32 comctl32)

# Optional official SDK fixture, never installed to the system plugin directory.
option(ASTRA_VST_BUILD_FIXTURE "Build official MDA test instruments" ON)
if(ASTRA_VST_BUILD_FIXTURE)
 set(vstsdk_VERSION 3.8.1)
 set(public_sdk_SOURCE_DIR "${SDK_ROOT}/public.sdk")
 set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
 set(SMTG_CREATE_MODULE_INFO OFF CACHE BOOL "" FORCE)
 set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
 function(smtg_target_setup_as_vst3_example target)
  smtg_target_configure_version_file(${target})
  target_compile_features(${target} PUBLIC cxx_std_17)
  target_link_libraries(${target} PRIVATE sdk)
 endfunction()
 add_subdirectory("${SDK_ROOT}/public.sdk/samples/vst/mda-vst3" "${CMAKE_BINARY_DIR}/mda-fixture")
endif()
